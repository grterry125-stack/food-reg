name: 自動更新食品法規資料
on:
  schedule:
    - cron: "7 */3 * * *"
  workflow_dispatch:
permissions:
  contents: write
jobs:
  update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: 衛福部與食藥署 RSS
        run: python fetch_mohw_rss.txt
        continue-on-error: true
      - name: 中醫藥司公告
        run: python fetch_dep_mohw.txt
        continue-on-error: true
      - name: 行政院公報卷期瀏覽
        run: python fetch_gazette_browse.txt
        continue-on-error: true
      - name: 農業部主管法規
        run: python fetch_moa_law.txt
        continue-on-error: true
      - name: 財政部與環境部法規
        run: python fetch_law_sites.txt
        continue-on-error: true
      - name: 全國法規資料庫最新訊息
        run: python fetch_moj_sitemap.txt
        continue-on-error: true
      - name: 合併資料
        run: python build_data.txt
      - name: 存回網站
        run: |
          git config user.name "auto-update"
          git config user.email "auto-update@users.noreply.github.com"
          git add data.json
          git diff --cached --quiet || (git commit -m "更新資料" && git push)

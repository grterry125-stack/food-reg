#!/usr/bin/env python3
"""把各來源抓到的公告合併成網站讀取的 data.json。
每個來源的結果各自獨立列出，不因相似或相同而合併；同一則公告出現在不同來源，就會各顯示一筆，
並標明來源與該來源自己的連結。累積歷史，最多保留 1500 則。"""
import json, re
from datetime import datetime, timezone, timedelta

OUT = "data.json"
FEEDS = {"食藥署公告": ("fda", "食藥署公告"),
         "農業部公告": ("moa", "農業部公告")}
SITES = {"mof": "財政部主管法規", "moenv": "環境部主管法規"}

def load(p):
    try:
        return json.load(open(p, encoding="utf-8"))
    except Exception:
        return {}

def norm_date(s):
    m = re.search(r"(\d{2,4})[-./](\d{1,2})[-./](\d{1,2})", s or "")
    if not m:
        return ""
    y, mo, d = int(m[1]), int(m[2]), int(m[3])
    if y < 1911:
        y += 1911
    return f"{y:04d}-{mo:02d}-{d:02d}"

old = load(OUT)
JUNK = ("系統訊息", "")
items = {i["uid"]: i for i in old.get("items", []) if i.get("uid") and i.get("k") and not str(i["uid"]).startswith("moj:gz-") and (i.get("title") or "").strip() not in JUNK and i.get("k") != "mohw"}
bad = set(old.get("moj_bad", []))
gazette_done = set(old.get("gazette_done", []))

def put(k, sn, id_, date, dept, title, url, dl=None):
    uid = f"{k}:{id_}"
    items[uid] = {"uid": uid, "k": k, "sn": sn, "date": date, "dept": dept, "title": title, "url": url, "dl": dl}

for i in load("out/mohw_items.json").get("all_new", []):
    if i.get("source_feed") not in FEEDS:
        continue
    k, sn = FEEDS[i["source_feed"]]
    put(k, sn, i["news_id"], i["publish_date"], i["agency"], i["title"], i["source_url"])

gz = load("out/gazette_items.json")
gazette_done |= set(gz.get("done_dates", []))
for i in gz.get("items", []):
    put("gazette", "行政院公報資訊網", i["id"], i["date"], i["dept"], i["title"], i["url"], i.get("deadline"))

for i in load("out/moalaw_items.json").get("items", []):
    put("moalaw", "農業部主管法規", i["id"], i["date"], "農業部", i["title"], i["url"], i.get("dl"))

for i in load("out/lawsites_items.json").get("items", []):
    put(i["site"], SITES.get(i["site"], i["site"]), i["id"], i["date"], i["dept"], i["title"], i["url"], i.get("dl"))

for i in load("moj_manual.json").get("items", []):      # 在自己電腦上執行後上傳的全國法規資料庫資料
    if (i.get("title") or "").strip() in JUNK:
        continue
    put("moj", "全國法規資料庫最新訊息", i["url"], norm_date(i.get("date")), "全國法規資料庫", i["title"], i["url"])

for i in load("out/dep_items.json").get("items", []):
    put("tcm", "中醫藥司公告訊息", i["id"], i["date"], "衛生福利部・中醫藥司", i["title"], i["url"])

moj = load("out/moj_news_food.json")
bad |= set(moj.get("bad", []))
if moj.get("cooldown_keep"):
    moj_cool = moj["cooldown_keep"]
elif moj.get("blocked"):
    moj_cool = (datetime.now(timezone.utc) + timedelta(hours=12)).strftime("%Y-%m-%dT%H:%M:%S")
else:
    moj_cool = ""
for i in moj.get("food_related", []):
    if (i.get("title") or "").strip() in JUNK:
        continue
    put("moj", "全國法規資料庫最新訊息", i["url"], norm_date(i.get("date")) or norm_date(i.get("lastmod")),
        "全國法規資料庫", i["title"], i["url"])

rows = sorted(items.values(), key=lambda x: (x.get("date") or ""), reverse=True)[:1500]
tw = datetime.now(timezone(timedelta(hours=8))).strftime("%Y-%m-%d %H:%M")
json.dump({"updated": tw, "schema": 2, "moj_bad": sorted(bad)[-3000:], "moj_cooldown_until": moj_cool, "gazette_done": sorted(gazette_done)[-400:], "items": rows}, open(OUT, "w", encoding="utf-8"), ensure_ascii=False, indent=1)
print(f"共 {len(rows)} 則，更新時間 {tw}")

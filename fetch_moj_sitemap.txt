#!/usr/bin/env python3
"""
全國法規資料庫（符合 robots.txt 的抓法）

該站 robots.txt 的規則是：全站禁止，只開放三個路徑：
  /News/NewsDetail.aspx（最新訊息內文）
  /LawClass/LawAll.aspx（法規全文頁）
  /SearchEngineSitemap.aspx（網站地圖）
所以不碰首頁 Index.aspx，而是：讀網站地圖 → 找出最新的訊息內文頁 → 逐頁讀取。
（網站地圖的實際格式我沒看過，若讀不到東西，請用 --debug 執行並把輸出貼給我。）

用法：python fetch_moj_sitemap.py            （最新訊息）
      python fetch_moj_sitemap.py --debug    （顯示細節）
輸出：out/moj_news_food.json
"""
import argparse, html, json, os, re, time, urllib.request
from datetime import datetime
from html.parser import HTMLParser
from urllib.parse import urlparse

BASE = "https://law.moj.gov.tw"
SITEMAP = BASE + "/SearchEngineSitemap.aspx"
UA = "food-reg-platform-test/0.1 (+https://github.com/grterry125-stack/food-reg)"   # 聯絡方式＝本專案的 GitHub 頁面
KEYWORDS = ["食品", "食安", "餐飲", "添加物", "農藥", "動物用藥", "動物藥品", "健康食品",
            "包裝飲用水", "屠宰", "乳品", "過敏原", "營養標示", "衛生標準", "衛生福利部"]

def get(url):
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(req, timeout=30) as r:
        return r.read().decode(r.headers.get_content_charset() or "utf-8", "replace")

# ---- robots.txt：用「最長符合」規則判斷（Python 內建的會被 Disallow: / 誤擋，所以自己寫）----
def load_rules():
    rules, on = [], False
    for line in get(BASE + "/robots.txt").splitlines():
        line = line.split("#")[0].strip()
        if not line or ":" not in line:
            continue
        k, v = [x.strip() for x in line.split(":", 1)]
        if k.lower() == "user-agent":
            on = (v == "*")
        elif on and k.lower() in ("allow", "disallow") and v:
            rules.append((k.lower(), v))
    return rules

def allowed(url, rules):
    u = urlparse(url); path = u.path + ("?" + u.query if u.query else "")
    best = None
    for kind, pat in rules:
        if path.startswith(pat) and (best is None or len(pat) > len(best[1])
                                     or (len(pat) == len(best[1]) and kind == "allow")):
            best = (kind, pat)
    return best is None or best[0] == "allow"

# ---- 網站地圖 ----
def sitemap_urls(url, rules, depth=0, debug=False):
    if not allowed(url, rules) or depth > 2:
        return []
    xml = get(url); out = []
    for blk in re.findall(r"<(url|sitemap)>(.*?)</\1>", xml, re.S):
        loc = re.search(r"<loc>(.*?)</loc>", blk[1], re.S)
        mod = re.search(r"<lastmod>(.*?)</lastmod>", blk[1], re.S)
        if not loc: continue
        u = html.unescape(loc.group(1).strip()); m = mod.group(1).strip() if mod else ""
        if blk[0] == "sitemap":
            out += sitemap_urls(u, rules, depth + 1, debug)
        else:
            out.append((u, m))
    if debug: print(f"地圖 {url} 共 {len(out)} 個網址")
    return out

class Text(HTMLParser):
    def __init__(s): super().__init__(); s.t=[]; s.skip=0; s.title=""; s._in=False
    def handle_starttag(s, tag, a):
        if tag in ("script","style","nav","footer","header"): s.skip += 1
        if tag == "title": s._in = True
    def handle_endtag(s, tag):
        if tag in ("script","style","nav","footer","header"): s.skip = max(0, s.skip-1)
        if tag == "title": s._in = False
    def handle_data(s, d):
        d = " ".join(d.split())
        if not d: return
        if s._in: s.title += d
        elif not s.skip: s.t.append(d)

def known_urls():
    """讀專案裡已有的 data.json，找出全國法規資料庫已蒐集過的網址，下次就不用重抓。"""
    try:
        d = json.load(open("data.json", encoding="utf-8"))
        return {x["url"] for x in d.get("items", []) if x.get("k") == "moj"} | set(d.get("moj_bad", []))
    except Exception:
        return set()

def msgid(u):
    m = re.search(r"msgid=(\d+)", u, re.I) or re.search(r"(\d+)\s*$", u)
    return int(m[1]) if m else 0

STATS = {"ok": 0, "sys": 0, "err": 0}
DATE_RE = r"(?:公\(發\)布日期|發文日期)[：: \n]*(\d{2,4}[-./]\d{1,2}[-./]\d{1,2})"

def read_item(u, tries=3):
    """讀一則訊息頁。回傳 (項目, 狀態)；狀態：ok／sys（系統訊息或查無資料）／err（連線失敗）"""
    try:
        p = Text(); p.feed(get(u, tries=tries))
    except Exception as e:
        print(f"[連線失敗] {u}\n       原因：{e}")
        STATS["err"] += 1
        return None, "err"
    body = "\n".join(p.t)
    title = re.sub(r"\s*[-－]\s*全國法規資料庫.*$", "", p.title or "").strip()
    if "系統訊息" in title or "系統訊息" in body[:30] or not title:
        print(f"[略過] {u}\n       頁面內容：{' '.join(body.split())[:160]}")
        STATS["sys"] += 1
        return None, "sys"
    STATS["ok"] += 1
    d = re.search(DATE_RE, body) or re.search(r"(\d{2,4}[-./]\d{1,2}[-./]\d{1,2})", body)
    return {"title": title, "url": u, "msgid": msgid(u), "date": d.group(1) if d else "", "body": body[:3000]}, "ok"

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--limit", type=int, default=30)      # 每次最多讀幾則網站地圖裡「新的」訊息
    ap.add_argument("--probe", type=int, default=40)      # 每次最多往後探幾個新編號（網站地圖更新比較慢，用來補最新的）
    ap.add_argument("--delay", type=float, default=5.0)
    ap.add_argument("--debug", action="store_true")
    a = ap.parse_args()

    rules = load_rules()
    if a.debug: print("robots 規則：", rules)
    urls = sitemap_urls(SITEMAP, rules, debug=a.debug)
    pairs = {u: m for u, m in urls if "NewsDetail.aspx" in u}
    by_id = sorted(pairs, key=msgid, reverse=True)                       # 編號越大越新
    by_mod = sorted(pairs, key=lambda u: pairs[u], reverse=True)         # 最近有更新的
    have = known_urls()
    todo, seen = [], set()
    for u in [x for pair in zip(by_id, by_mod) for x in pair]:          # 兩種排序交錯取
        if u not in have and u not in seen:
            seen.add(u); todo.append(u)
    todo = todo[:a.limit]
    print(f"網站地圖共 {len(by_id)} 則訊息，已蒐集 {len(have)} 則，本次讀取新的 {len(todo)} 則")

    items, bad, streak, estreak = [], [], 0, 0
    aborted = False
    for u in todo:
        if not allowed(u, rules): continue
        time.sleep(a.delay)
        it, st = read_item(u)
        if st == "err":
            estreak += 1
            if estreak >= 3:
                print("連續 3 次連線失敗，疑似被網站暫時限制，本次提早結束，稍後會自動再試。")
                aborted = True; break
            continue
        estreak = 0
        if st == "ok":
            streak = 0; items.append(it)
        elif st == "sys":
            streak += 1
            if streak >= 3:
                print("連續 3 頁都是「系統訊息」，疑似被網站暫時限制，本次提早結束，稍後會自動再試。")
                bad = []; aborted = True; break
            bad.append(u)

    # 往後探新編號：網站地圖比官網慢，編號是連續遞增的，所以從目前最大的有效編號往後一個一個讀
    probed = 0
    if not aborted and a.probe > 0:
        top = max([msgid(u) for u in have] + [x["msgid"] for x in items] + [0])
        miss, n, errs = 0, top, 0
        while probed < a.probe and miss < 8 and top:
            n += 1; probed += 1
            u = f"{BASE}/News/NewsDetail.aspx?msgid={n}"
            if not allowed(u, rules): break
            time.sleep(a.delay)
            it, st = read_item(u, tries=1)
            errs = errs + 1 if st == "err" else 0
            if errs >= 3:
                print("往後探時連續 3 次連線失敗，提早結束。"); break
            if st == "ok":
                miss = 0; items.append(it)
            else:
                miss += 1
        print(f"往後探 {probed} 個新編號（從 {top + 1} 開始），讀到 {sum(1 for x in items if x['msgid'] > top)} 則")

    os.makedirs("out", exist_ok=True)
    json.dump({"fetched_at": datetime.now().isoformat(timespec="seconds"),
               "source": "全國法規資料庫（法務部）最新訊息", "total": len(items), "food_related": items, "bad": bad},
              open("out/moj_news_food.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    print(f"讀取 {len(items)} 則 → out/moj_news_food.json（成功 {STATS['ok']}、系統訊息 {STATS['sys']}、連線失敗 {STATS['err']}）")
    if STATS["err"] and not STATS["ok"]:
        print("全部都連線失敗，疑似被網站暫時限制，請等幾個小時、不要手動連續執行。")
    for i in sorted(items, key=lambda x: x["msgid"], reverse=True)[:10]: print(" -", i["msgid"], i["date"], i["title"][:36])

if __name__ == "__main__":
    main()

#!/usr/bin/env python3
"""
衛福部官方 RSS → 食品法規候選公告（只用 Python 標準函式庫）

來源：https://www.mohw.gov.tw/cp-2661-6125-1.html 列出的官方 RSS
  公告訊息 rss-18-1.html、焦點新聞 rss-16-1.html、活動訊息 rss-101-1.html
用法：python fetch_mohw_rss.py
輸出：out/mohw_items.json（本次候選，格式對應網站的資料欄位）
      out/seen_ids.json（已處理過的 NewsID，用來只挑出新公告）
請依官方 ttl（約 20 分鐘）控制頻率，建議每 1～3 小時執行一次。
"""
import html, json, os, re, urllib.request
import xml.etree.ElementTree as ET
from datetime import datetime
from email.utils import parsedate_to_datetime

FEEDS = {
    "衛福部公告訊息": "https://www.mohw.gov.tw/rss-18-1.html",
    "食藥署公告": "https://www.fda.gov.tw/tc/rssAnnouncement.ashx",
}
UA = "food-reg-platform-test/0.1 (contact: your-email@example.com)"   # ← 改成你的聯絡方式

FOOD = ["食品", "食安", "餐飲", "添加物", "農藥", "動物用藥", "健康食品", "包裝飲用水",
        "屠宰", "乳品", "過敏原", "營養標示", "衛生標準", "進口查驗"]
REG = ["預告", "修正", "訂定", "廢止", "發布", "公告", "令", "辦法", "標準", "規定", "草案", "法"]
CATS = {  # 12 大分類關鍵字（簡易規則，正式版交給 AI 判讀）
    "食品添加物": ["添加物"], "食品標示管理": ["標示", "過敏原", "宣稱"],
    "農藥及動物用藥": ["農藥", "動物用藥", "殘留"], "食品輸入管理": ["輸入", "進口", "邊境", "查驗"],
    "食品追溯追蹤": ["追溯", "追蹤"], "GHP／HACCP": ["GHP", "HACCP", "良好衛生規範"],
    "食品檢驗管理": ["檢驗", "抽驗"], "食品衛生標準": ["衛生標準", "污染物", "微生物"],
    "食品業者管理": ["登錄", "業者"], "餐飲及其他管理": ["餐飲", "食品中毒"],
    "食品安全基本法規": ["食品安全衛生管理法"], "食品原料管理": ["原料"],
}

def strip_html(s):
    return " ".join(html.unescape(re.sub(r"<[^>]+>", " ", s or "")).replace("]]>", " ").split())

def change_type(title):
    for k, v in (("預告", "預告"), ("廢止", "廢止"), ("修正", "修正"), ("訂定", "新增")):
        if k in title:
            return v
    return "公告"

def guess_category(text):
    for cat, keys in CATS.items():
        if any(k in text for k in keys):
            return cat
    return "待分類"

def fetch(url):
    req = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(req, timeout=30) as r:
        return r.read()

def parse(name, xml_bytes):
    root = ET.fromstring(xml_bytes)
    for it in root.iter("item"):
        g = lambda t: (it.findtext(t) or "").strip()
        title, desc = g("title"), strip_html(g("description"))
        try:
            pub = parsedate_to_datetime(g("pubDate")).date().isoformat()
        except Exception:
            pub = ""
        yield {
            "news_id": g("NewsID") or re.sub(r"&utm_.*", "", g("link")),
            "title": title, "publish_date": pub,
            "agency": "衛生福利部・" + (g("DeptName") or ("食品藥物管理署" if "fda.gov.tw" in g("link") else "")),
            "source_feed": name, "source_url": re.sub(r"&utm_.*", "", g("link")) or g("source"),
            "summary_raw": desc[:500],
            "public_end_date": g("PublicEndDate") or None,
            "attachments": [{"name": f.get("name"), "url": f.get("url")} for f in it.iter("File")],
            "change_type": change_type(title),
            "category": guess_category(title + desc),
            "is_food_related": any(k in (title + desc).replace("食品藥物管理署", "") for k in FOOD),
            "is_regulation_like": any(k in title for k in REG),
            "review_status": "待人工覆核",
        }

def main():
    os.makedirs("out", exist_ok=True)
    seen_path = "out/seen_ids.json"
    seen = set(json.load(open(seen_path))) if os.path.exists(seen_path) else set()
    allitems, new = [], []
    for name, url in FEEDS.items():
        try:
            for item in parse(name, fetch(url)):
                allitems.append(item)
                if item["news_id"] not in seen:
                    new.append(item)
        except Exception as e:
            print(f"[{name}] 失敗：{e}")
    hits = [i for i in new if i["is_food_related"]]
    json.dump({"fetched_at": datetime.now().isoformat(timespec="seconds"),
               "source": "衛生福利部官方 RSS", "total": len(allitems),
               "new": len(new), "food_candidates": hits, "all_new": new},
              open("out/mohw_items.json", "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    json.dump(sorted(seen | {i["news_id"] for i in allitems}), open(seen_path, "w"))
    print(f"讀取 {len(allitems)} 則，新公告 {len(new)} 則，食品相關候選 {len(hits)} 則")
    for i in hits[:10]:
        print(" -", i["publish_date"], i["change_type"], i["title"][:45])

if __name__ == "__main__":
    main()
